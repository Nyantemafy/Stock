package Test;

import java.math.BigDecimal;

public class Test {     
    public static BigDecimal ajouterStock(BigDecimal stock, BigDecimal entree) {
        return stock.add(entree);
    }

    public static void main(String[] args) {
        BigDecimal stockInitial = new BigDecimal("100.00");
        System.out.println("stock initial: " + stockInitial);

        BigDecimal stockEnter = new BigDecimal("30.50");
        System.out.println("entre: " + stockEnter);
        BigDecimal nouveauStock = ajouterStock(stockInitial, stockEnter);
        System.out.println("nouveau stock : " + nouveauStock);
    }
}

package Test;

import java.math.BigDecimal;

public class Test {

    public static boolean stockDisponible(BigDecimal stock, BigDecimal sortie) {
        if (stock.compareTo(sortie) >= 0) {
            return true;
        } else {
            return false;
        }
    }

    public static void main(String[] args) {
        BigDecimal stock = new BigDecimal("100.00");
        BigDecimal stockSorti = new BigDecimal("30.00");

        boolean estDisponible = stockDisponible(stock, stockSorti);
        if (estDisponible) {
            System.out.println("Stock suffisant");
        } else {
            System.out.println("Stock insuffisant");
        }
    }
}

package dao;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

public class ConnexionPostgres {
    private static final String URL = "jdbc:postgresql://localhost:5432/stock";
    private static final String USER = "postgres";
    private static final String PASSWORD = "antema";

    public static Connection ouvrir() throws SQLException {
        return DriverManager.getConnection(URL, USER, PASSWORD);
    }
}
